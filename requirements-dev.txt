# Development and CI tooling. The plugin itself has no runtime dependency that is
# not vendored - see NOTICE.
#
# Pinned exactly, not with `>=`. A floating lint or test tool means CI can go red
# on a commit that changed nothing, and a contributor's local run can disagree
# with the runner for a week before anybody works out why. Bump these on purpose,
# in their own commit, with the failure they fix or the release they follow.
#
# pytest is split on the Python version because the matrix has to keep working:
# 9.x requires Python >= 3.10 and the floor here is 3.9 (OpenPLi 9), so 3.9 gets
# the last 8.x. Everything else installs on every version in the matrix.
ruff==0.16.7
pyflakes==3.4.0

# Test-only. The discovery payloads carry Jinja templates that Home Assistant
# renders, and comparing them as strings let a timestamp template ship with the
# offset written twice. The tests render every one of them instead, through
# tests/hatemplate.py; nothing in src/ imports this and the IPK does not carry it.
jinja2==3.1.6

# Test-only as well. tools/check-workflows.py reads the workflows with a small YAML reader of its
# own, because the signing job's Python has nothing installed; a test holds that reader to PyYAML's
# reading of every workflow in the repository.
pyyaml==6.0.3
pytest==9.1.1 ; python_version >= "3.10"
pytest==8.4.2 ; python_version < "3.10"
